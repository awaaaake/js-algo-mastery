### 문제풀이

최소신장 트리 : MST <br/>
-> 최소비용으로 사이클없이, 모든 정점을 연결하는 트리<br/>
- 크루스칼 알고리즘(간선이 적을 때)<br/>
- 프림 알고리즘<br/>

### 크루스칼 알고리즘의 핵심
- `유니온 파인드`<br/>
1. 파인드 : x의 루트 노드 찾기<br/>
2. 루트 노드가 다르면 같은 집합으로 합친다 = 루트 노드 끼리 연결한다.<br/>

```js
const [N, M] = input[0];
const edges = [];
let parent = Array.from({ length: N + 1 }, (_, i) => i); //자기자신을 부모노드(루트)로 설정

const find = (x) => {
  //루트 노드를 찾는 함수
  if (parent[x] === x) return x;
  return (parent[x] = find(parent[x])); //경로압축
};

const union = (a, b) => {
  //루트 노드끼리 합치는 함수
  let rootA = find(a);
  let rootB = find(b);
  if (rootA !== rootB) {
    a < b ? (parent[rootB] = rootA) : (parent[rootA] = rootB);
  }
};

for (let i = 1; i <= M; i++) {
  let [a, b, cost] = input[i];
  edges.push([cost, a, b]);
}

edges.sort((a, b) => a[0] - b[0]); //간선비용을 오름차순으로 정렬

let totalCost = 0;
let maxEdge = 0;
let count = 0;

for (let [cost, a, b] of edges) {
  if (find(a) !== find(b)) {
    union(a, b);
    totalCost += cost;
    maxEdge = cost;
    count++;
    if (count === N - 1) break; //MST 완성성
  }
}
console.log(totalCost - maxEdge);
```
