### 문제풀이

1. DFS/BFS
- 인접노드 정보를 graph에 저장<br/>
- 0번노드부터 순회하면서 dfs를 실행(만약 방문하지 않았다면 count++ : 새로운 네트워크)<br/>
    - 이웃한 모든 노드들을 방문한다<br/>
    - 특정 노드를 한번 방문할 때 그 네트워크의 모든 노드들을 방문하게 된다.<br/>
```js
function solution(n, computers) {
  //dfs
  const graph = Array.from({ length: n }, () => []);
  for (let i = 0; i < n; i++) {
    for (let j = 0; j < n; j++) {
      if (computers[i][j] === 1 && i !== j) graph[i].push(j);
    }
  }

  let count = 0;
  const visited = new Array(n).fill(false);

  const dfs = (node) => {
    visited[node] = true;

    for (let next of graph[node]) {
      if (!visited[next]) {
        dfs(next);
      }
    }
  };

  for (let i = 0; i < n; i++) {
    if (!visited[i]) {
      dfs(i);
      count++;
    }
  }

  return count;
}
```

2. union-find<br/>
- computers의 연결된 노드끼리 union<br/>
- 0번째 노드부터 find(x)를 해야 -> 정확한 root가 나온다.(parent[x]가 x의 root가 아닐 수 있음)<br/>
```js
function solution(n, computers) {
  const parent = new Array(n).fill().map((_, i) => i);
  const find = (x) => {
    if (parent[x] === x) return x;
    return (parent[x] = find(parent[x]));
  };

  const union = (a, b) => {
    let rootA = find(a);
    let rootB = find(b);
    if (rootA !== rootB) {
      a < b ? (parent[rootB] = rootA) : (parent[rootA] = rootB);
    }
  };

  for (let i = 0; i < n; i++) {
    for (let j = 0; j < n; j++) {
      if (computers[i][j]) union(i, j);
    }
  }

  let count = 0;
  const groups = new Set();
  for (let i = 0; i < n; i++) {
    let root = find(i);
    if (!groups.has(root)) {
      count++;
      groups.add(root);
    }
  }
  return count;
}
```
