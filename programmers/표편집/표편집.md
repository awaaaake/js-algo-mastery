### 문제풀이

1차 풀이 방식
```js
function solution(n, k, cmd) {
    let curr = k;
    const checked = new Array(n).fill(true);
    const lastDeleted = [];
    
    for(let c of cmd) {ㅎ
        const [char, num] = c.split(' ');
        if(char === "U") {
            let cnt = Number(num);
            while(cnt>0) {
                curr--;
                if(checked[curr]) {
                    cnt--;
                }
            }
        } else if(char === "D") {
            let cnt = Number(num);
            while(cnt>0) {
                curr++;
                if(checked[curr]) {
                    cnt--;
                }
            }
        } else if(char === "C") {
            checked[curr] = false;
            lastDeleted.push(curr);
            if(curr === n - 1) {
                while(!checked[curr]) {
                    curr--;
                }
            } else {
                while(!checked[curr]) {
                    curr++;
                }
            }
        } else if(char === "Z") {
            let lst = lastDeleted.pop();
            checked[lst] = true;
        }
    }
    let result = "";
    for(let index of checked) {
        result += index ? "O" : "X";
    }
    return result;
}
```
- 시간초과 발생
### 이유?
U와 D명령어를 처리하는 과정에서 반복문을 돌며 매번 유효한(삭제되지않은 행)행을 찾기 떄문이다. 삭제된행을 건너뛰는 로직이 있어, 행이 많이 삭제될수록 효율이 떨어진다.<BR/>

### 시간초과를 어떻게 해결할까?
💡 연결리스트 활용<BR/>
삭제된 행을 건너뛰는 과정을 빠르게 처리할 수 있도록 자료구조를 개선해야한다<BR/>
U와 D가 특정 칸 만큼 바로 이동할 수 있도록 연결리스트구조를 사용할 수 있다.<BR/>

### 개선된 풀이                  
 ```js
 const Node = function (index, prevNode) {
    this.index = index;
    this.prev= prevNode;
    this.next;
};

function solution(n, k, cmd) {
    let root = new Node(0);
    let currNode = root;
    let prevNode = root;
    
    for(let i=1; i<n; i++) {
        const newNode = new Node(i, prevNode);
        prevNode.next = newNode;
        prevNode = newNode;
        
        if(i===k) {
            currNode = newNode;
        }
    }
    
    const checked = new Array(n).fill(true);
    const lastDeleted = [];
    
    for(let c of cmd) {
        const [char, num] = c.split(' ');
        if(char === "U") {
            let i=0;
            while(i<Number(num) && currNode.prev) {
                currNode = currNode.prev;
                i++;
            }
        } else if(char === "D") {
            let i=0;
            while(i<Number(num) && currNode.next) {
                currNode = currNode.next;
                i++;
            }
        } else if(char === "C") {
            lastDeleted.push(currNode);
            const prev = currNode.prev;
            const next = currNode.next;
            if(prev && next) {
                prev.next=next;
                next.prev=prev;
                currNode=next;
            } else if(prev) {//마지막 행이면
                prev.next=null;
                currNode=prev;
            } else if(next) {//첫번째 행이면
                next.prev=null;
                currNode=next;
            }
        } else if(char === "Z") {
            let lst = lastDeleted.pop();
            const prev=lst.prev;
            const next=lst.next;
            if(prev) {
                prev.next=lst;
            }
            if(next) {
                next.prev=lst;
            }
        }
    }
    
    let result = new Array(n).fill("O");
    for(let node of lastDeleted) {
        result[node.index]="X";
    }
    
    return result.join("");
}
```
- Node 생성자 함수로 node객체를 생성한다.
    - 해당 노드의 인덱스, 이전노드, 다음노드를 정의한다.
- 양방향 연결리스트를 구현하여 각 노드가 이전노드와 다음 노드를 가리키도록 하고 있다.
- 연결리스트에 노드를 추가하거나 삭제할 때, prev와 next를 수정하면 된다.
> 💡생성자 함수 : js에서 객체를 생성하는 방법
