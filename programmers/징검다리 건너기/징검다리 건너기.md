### 문제풀이

### 처음 풀이 : 선형탐색

```js
function solution(stones, k) {
  let answer = 0;

  for (let num = 1; num < Math.max(...stones); num++) {
    let isPossible = true;
    const temp = stones.map((el) => (el - num < 0 ? 0 : el - num));
    let curr = 0;
    let chance = k - 1; //연속가능한 0의 개수
    while (curr < stones.length) {
      if (temp[curr] === 0) {
        chance--;
        if (chance < 0) {
          isPossible = false;
          break;
        }
      } else {
        chance = k - 1; //돌하나 밟으면 chance 리셋
      }
      curr++;
    }

    if (isPossible) answer = num + 1;
  }
  return answer;
}
```

- 건너뛸 수 있는 최대 인원을 num=1부터 1씩증가하면서 찾고있다. -> 시간복잡도 : 선형탐색: `O(M*N)` (M: Math.max(...stones), N: 돌의 개수)

### 이분탐색을 사용하는 방식

- `가능한 인원 수의 범위`를 `이분탐색`으로 줄여나가면서 `돌을 건널 수 있는지` 확인한다.
- 각 경우에 대해 `돌을 전부 순회하며 건너뛰기를 시도`하면 - 시간복잡도 : `O(N*logM)`
  > ⚠️ right를 설정할 때, Math.max(...stones)와 같이 배열의 최대값을 매번 계산하는 부분이 문제다. 배열의 크기가 매우 클 경우, 매번 최대값을 계산하면서 비효율적으로 메모리를 사용하게 되고, 이는 런타임 에러를 유발할 수 있다.<br/>
  > 따라서 문제 요구사항 `stones 배열 각 원소들의 값은 1 이상 200,000,000 이하인 자연수입니다.`를 기준으로 right를 200,000,000로 설정한다.<br/>
  > ⭐ 결론 : `이분탐색의 right`는 `문제의 요구사항 조건(범위)를 확인`해 최댓값으로 설정한다.

```js
function solution(stones, k) {
  let left = 1;
  let right = 200000000; //징검다리를 건널 수 있는 최대 인원
  let answer = 0;

  while (left <= right) {
    let mid = Math.floor((left + right) / 2);
    let zeros = 0; //연속된 0의 개수
    let isPossible = true;

    for (let st of stones) {
      //가능 여부를 한 사람이 건너뛴, 그 다음 징검다리 상태에서부터 판별함
      if (st - mid <= 0) {
        //mid만큼 건너뛰었다고 생각하고 판별 -> mid+1명도 가능한가를 판별
        zeros++;
        if (zeros >= k) {
          isPossible = false;
          break;
        }
      } else {
        zeros = 0;
      }
    }

    if (isPossible) {
      answer = Math.max(mid + 1, answer); //현재 mid명 까지 건넜을때 연속되는 0의 수가 k보다 작다는 것은, 그 다음 친구까지는 무조건 건널 수 있음
      left = mid + 1;
    } else {
      right = mid - 1;
    }
  }

  return answer;
}
```
